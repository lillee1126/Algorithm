## 题目与题解

[题目传送门](https://leetcode.cn/problems/longest-increasing-subsequence/)
```
class Solution {
    int d[2510];
public:
    int lengthOfLIS(vector<int>& nums) {
        int n=nums.size();
        int len=1;
        if(n==0){
            return 0;
        }
        d[len]=nums[0];
        for(int i=1;i<n;i++){
            if(nums[i]>d[len]){
                len++;
                d[len]=nums[i];
            }
            else{
                int l=1,r=len,pos=0;
                while(l<=r){
                    int mid=(l+r)/2;
                    if(d[mid]<nums[i]){
                        pos=mid;
                        l=mid+1;
                    }
                    else{
                        r=mid-1;
                    }
                }
                d[pos+1]=nums[i];
            }
        }
        return len;
    }
};
```
