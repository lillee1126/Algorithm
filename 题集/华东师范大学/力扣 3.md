class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int n=s.size();
        int ans=0;
        if(s.size()<=1){
            return n;
        }
        map<char,int> m;
        int l=0,r=0;
        while(r<n){
            if(m.count(s[r])!=1){
                ans=max(ans,r-l+1);
                m[s[r]]=r;
                r++;
            }
            else{
                if(m[s[r]]>=l){
                    l=m[s[r]]+1;
                }
                else{
                    ans=max(ans,r-l+1);
                }
                
                m[s[r]]=r;
                r++;
            }
        }
        return ans;
    }
};
