//(i−1)a+(n−i)b=(n−1)b+(i−1)(a−b)
#include <bits/stdc++.h>
#define int long long
#define endl '\n'
using namespace std;
int n;
const int N=1e5+10;
pair<int,int> a[N];
bool f(const pair<int,int>& aa,const pair<int,int>& bb){
    return aa.first-aa.second>bb.first-bb.second;
}
void solve(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i].first>>a[i].second;
    }
    sort(a+1,a+1+n,f);
    int ans=0;
    for(int i=1;i<=n;i++){
        ans+=(n-1)*a[i].second;
        ans+=(i-1)*(a[i].first-a[i].second);
    }
    cout<<ans;
}
signed main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
	cout.tie(0);
	solve();
	return 0;
}
