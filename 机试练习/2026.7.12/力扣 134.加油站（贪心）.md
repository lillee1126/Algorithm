[题目传送门](https://leetcode.cn/problems/gas-station/description/)
```
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int n=gas.size();
        int sa=0,sb=0;
        for(int i=0;i<n;i++){
            sa+=gas[i];
            sb+=cost[i];
        }
        if(sb>sa){
            return -1;
        }
        int cur=0;
        int ans=-1;
        int i=0;
        while(i<n){
            cur+=gas[i]-cost[i];
            if(cur<0){
                ans=i;
                cur=0;
            }
            i++;
        }
        return ans+1;
    }
};
```
