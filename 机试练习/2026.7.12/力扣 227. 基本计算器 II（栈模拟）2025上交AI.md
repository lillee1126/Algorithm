[题目传送门](https://leetcode.cn/problems/basic-calculator-ii/)
```
class Solution {
public:
    int calculate(string s) {
        stack<long long> sz;  // 数字栈
        stack<char> fh;       // 符号栈

        int i = 0;
        int n = s.size();

        while (i < n) {
            // 跳过空格
            if (s[i] == ' ') {
                i++;
                continue;
            }

            // 读取一个完整数字
            if (s[i] >= '0' && s[i] <= '9') {
                long long a = 0;

                while (i < n && s[i] >= '0' && s[i] <= '9') {
                    a = a * 10 + (s[i] - '0');
                    i++;
                }

                sz.push(a);
            } else {
                char c = s[i];
                i++;

                // 跳过运算符后面的空格
                while (i < n && s[i] == ' ') {
                    i++;
                }

                if (c == '*' || c == '/') {
                    long long a = 0;

                    // 这里只能在当前字符是数字时读取
                    while (i < n && s[i] >= '0' && s[i] <= '9') {
                        a = a * 10 + (s[i] - '0');
                        i++;
                    }

                    long long left = sz.top();
                    sz.pop();

                    if (c == '*') {
                        sz.push(left * a);
                    } else {
                        sz.push(left / a);
                    }
                } else {
                    // 加减法最后处理
                    fh.push(c);
                }
            }
        }

        // 反转数字栈和符号栈，保证加减法从左向右运算
        stack<long long> nums;
        stack<char> ops;

        while (!sz.empty()) {
            nums.push(sz.top());
            sz.pop();
        }

        while (!fh.empty()) {
            ops.push(fh.top());
            fh.pop();
        }

        long long ans = nums.top();
        nums.pop();

        while (!ops.empty()) {
            char op = ops.top();
            ops.pop();

            long long num = nums.top();
            nums.pop();

            if (op == '+') {
                ans += num;
            } else {
                ans -= num;
            }
        }

        return static_cast<int>(ans);
    }
};
```
